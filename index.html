precision mediump float; uniform float truthAura; uniform float mythAura; uniform float driftAura; uniform float tensorAura; uniform float orbitAura; uniform float pulseAura; uniform float waveAura; uniform float fieldAura; uniform vec3 elementAura; varying vec3 vPos; float waveField(vec3 p, float w) { return sin(p.x * 3.14 + w * 6.28) * cos(p.y * 3.14 + w * 3.14); } vec3 chromaticRift(vec3 c, float f) { return vec3( c.r + f * 0.25, c.g + f * 0.10, c.b + f * 0.35 ); } vec3 truthBloom(vec3 base, float t) { return base * (1.0 + t * 1.5); } vec3 mythFog(float m) { return vec3(0.05, 0.05, 0.1) * m; } vec3 fieldDistortion(float f) { return vec3(f * 0.4, f * 0.2, f * 0.1); } vec3 pulseHarmonics(float p) { return vec3( sin(p * 6.28), sin(p * 3.14), sin(p * 9.42) ) * 0.3; } vec3 orbitShift(float o) { return vec3( sin(o), sin(o * 0.5), sin(o * 0.25) ); } void main() { vec3 base = elementAura; float wave = waveField(vPos, waveAura); vec3 color = truthBloom(base, truthAura) + pulseHarmonics(pulseAura) + orbitShift(orbitAura) + fieldDistortion(fieldAura) + vec3(wave) - mythFog(mythAura); color = chromaticRift(color, fieldAura); gl_FragColor = vec4(color, 1.0); }
