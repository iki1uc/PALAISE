precision mediump float; uniform float truthAura; uniform float mythAura; uniform float driftAura; uniform float tensorAura; uniform float orbitAura; uniform float pulseAura; uniform float waveAura; uniform float fieldAura; uniform vec3 elementAura; uniform float time; varying vec3 vPos; /* Fraktale Tiefenstruktur */ float fractal(vec3 p) { float v = 0.0; float s = 1.0; for (int i = 0; i < 5; i++) { v += abs(sin(p.x * s + p.y * s + p.z * s + time * 0.3)); s *= 1.7; } return v; } /* Chromatische Aura‑Brechung */ vec3 chromatic(vec3 c, float f) { return vec3( c.r + f * 0.22, c.g + f * 0.08, c.b + f * 0.31 ); } /* Organische Wellen */ float waveField(vec3 p, float w) { return sin(p.x * 3.14 + w * 6.28 + time * 0.8) * cos(p.y * 3.14 + w * 3.14 + time * 0.4); } /* Wahrheit → Bloom */ vec3 truthBloom(vec3 base, float t) { return base * (1.0 + t * 1.8); } /* Irrtum → Nebel */ vec3 mythFog(float m) { return vec3(0.05, 0.05, 0.1) * m; } /* Puls → harmonische Schwingung */ vec3 pulseHarmonics(float p) { return vec3( sin(p * 6.28 + time), sin(p * 3.14 + time * 0.5), sin(p * 9.42 + time * 1.5) ) * 0.3; } /* Orbit → Farbrotation */ vec3 orbitShift(float o) { return vec3( sin(o + time * 0.4), sin(o * 0.5 + time * 0.2), sin(o * 0.25 + time * 0.1) ); } /* Feld → Raumverzerrung */ vec3 fieldDistortion(float f) { return vec3(f * 0.4, f * 0.2, f * 0.1); } void main() { vec3 base = elementAura; float wave = waveField(vPos, waveAura); float fract = fractal(vPos * (1.0 + fieldAura)); vec3 color = truthBloom(base, truthAura) + pulseHarmonics(pulseAura) + orbitShift(orbitAura) + fieldDistortion(fieldAura) + vec3(wave * 0.4 + fract * 0.2) - mythFog(mythAura); color = chromatic(color, fieldAura); gl_FragColor = vec4(color, 1.0); }
